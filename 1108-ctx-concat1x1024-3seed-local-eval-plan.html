下载与校验脚本放在日志目录 v1-store/logs/ctx-concat1x1024-3seed-1108/fetch.sh（运维脚本，不进仓库），在 tmux evl-cc-fetch 里按 1107 模板带 tee 和 EXIT_CODE= 运行：

B=https://huggingface.co/buckets/HongzeFu/robomme-vla-orig100-ctx32x4x4-motion-s2concat1x1024-60k-v1/resolve
R=/scratch/hongze/robomme_policy_learning_MotionJEPA/v1-store/train-runs/mme_vla_suite_orig16_b128_60k/orig100-ctx32x4x4-motion-s2concat1x1024-b128-60k
P=$R/59999.partial; mkdir -p $P
curl -sfL $B/manifests/step-59999.json -o $P/../manifest-59999.json
jq -r '.files | keys[]' $R/manifest-59999.json \
  | xargs -P 8 -I{} sh -c 'mkdir -p "$(dirname "'$P'/${1#59999/}")"; curl -sfL --retry 3 "'$B'/$1" -o "'$P'/${1#59999/}"' _ {}
jq -r '.files | to_entries[] | "\(.value.sha256)  \(.key | ltrimstr("59999/"))"' $R/manifest-59999.json > $P.SHA256SUMS
(cd $P && sha256sum -c --quiet $P.SHA256SUMS) && [ "$(find $P -type f | wc -l)" = 38 ] \
  && mv $P $R/59999 && echo "VLA_VERIFY=PASS step=59999 files=38 sha_mismatch=0" || echo "VLA_VERIFY=FAIL"
